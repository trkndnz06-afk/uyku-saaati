"""YouTube bağlantısı: bir kez yetkilendirme ve video yükleme."""
import os, sys, json, time
from pathlib import Path

KOK = Path(__file__).resolve().parent
SCOPES = ["https://www.googleapis.com/auth/youtube.upload", "https://www.googleapis.com/auth/youtube.readonly",
          "https://www.googleapis.com/auth/youtube.force-ssl"]
TOKEN = KOK / "token.json"; SECRET = KOK / "client_secret.json"


def creds():
    from google.oauth2.credentials import Credentials
    from google.auth.transport.requests import Request
    if not TOKEN.exists(): raise RuntimeError("YouTube bağlı değil: önce 'YouTube'a bağlan.command' dosyasını çalıştır.")
    c = Credentials.from_authorized_user_file(str(TOKEN), SCOPES)
    if not c.valid:
        c.refresh(Request()); TOKEN.write_text(c.to_json())
    return c


def service():
    from googleapiclient.discovery import build
    return build("youtube", "v3", credentials=creds(), cache_discovery=False)


def authorize():
    from google_auth_oauthlib.flow import InstalledAppFlow
    if not SECRET.exists():
        adaylar = sorted((Path.home() / "Downloads").glob("client_secret*.json"), key=lambda p: p.stat().st_mtime)
        if adaylar:
            import shutil; shutil.copy(adaylar[-1], SECRET); print("İndirilenler'deki", adaylar[-1].name, "kullanıldı.")
    if not SECRET.exists():
        print(f"client_secret.json bulunamadı. Google Cloud'dan indirdiğin dosyayı şu klasöre koy:\n{KOK}"); sys.exit(1)
    flow = InstalledAppFlow.from_client_secrets_file(str(SECRET), SCOPES)
    c = flow.run_local_server(port=0, prompt="consent", access_type="offline")
    TOKEN.write_text(c.to_json())
    ch = service().channels().list(part="snippet", mine=True).execute().get("items", [])
    print("Bağlandı. Kanal:", ch[0]["snippet"]["title"] if ch else "(kanal bulunamadı)")


def upload(video, thumb, meta, privacy="public", category="10", publish_at=None, log=print):
    from googleapiclient.http import MediaFileUpload
    from googleapiclient.errors import HttpError
    yt = service()
    status = {"privacyStatus": privacy, "selfDeclaredMadeForKids": False, "containsSyntheticMedia": False}
    if publish_at: status.update(privacyStatus="private", publishAt=publish_at)
    body = {"snippet": {"title": meta["title"], "description": meta["description"], "tags": meta["tags"], "categoryId": category,
                        "defaultLanguage": "en", "defaultAudioLanguage": "zxx"}, "status": status}
    media = MediaFileUpload(str(video), chunksize=64 * 1024 * 1024, resumable=True, mimetype="video/mp4")
    req = yt.videos().insert(part="snippet,status", body=body, media_body=media)
    resp = None; tries = 0
    while resp is None:
        try:
            st, resp = req.next_chunk()
            if st: log(f"yükleniyor %{int(st.progress()*100)}")
        except HttpError as e:
            if e.resp.status in (500, 502, 503, 504) and tries < 8:
                tries += 1; time.sleep(2 ** tries); continue
            raise
        except (ConnectionError, TimeoutError, OSError):
            if tries < 8: tries += 1; time.sleep(2 ** tries); continue
            raise
    vid = resp["id"]; log(f"yüklendi: https://youtu.be/{vid}")
    try:
        yt.thumbnails().set(videoId=vid, media_body=MediaFileUpload(str(thumb), mimetype="image/jpeg")).execute()
        log("thumbnail eklendi")
    except HttpError as e:
        log(f"thumbnail eklenemedi (kanalın telefonla doğrulanması gerekebilir): {e}")
    try:
        real = yt.videos().list(part="status", id=vid).execute()["items"][0]["status"]["privacyStatus"]
        if real != status["privacyStatus"]:
            log(f"UYARI: video '{real}' durumunda kaldı. Google Cloud projesi YouTube denetiminden geçene kadar API ile yüklenen videolar gizli kalır.")
    except Exception: pass
    return vid


if __name__ == "__main__" and len(sys.argv) > 1 and sys.argv[1] == "yetkilendir":
    authorize()
